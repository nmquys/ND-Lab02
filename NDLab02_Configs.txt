! ==============================================================
! ND-Lab02 – Advanced OSPF Area 0 / Redundant WAN
! Cấu hình sẵn để dán vào CLI từng thiết bị trong Packet Tracer
! Thiết bị: Router 1841 (R1,R3: 2×WIC-2T; R2,R4: 1×WIC-2T), Switch 2960
! clock rate chỉ có trên đầu DCE (R1: S0/0/0,S0/0/1,S0/1/0; R2: S0/0/1; R4: S0/0/1)
! Cáp: Serial DCE (bấm router DCE trước); Ethernet: Copper Straight-Through
! ==============================================================

!============================== R1 ==============================
enable
configure terminal
hostname R1
no ip domain-lookup
interface FastEthernet0/0
 description TO-SW1-LAN1
 ip address 192.168.1.1 255.255.255.0
 no shutdown
interface Serial0/0/0
 description WAN1-TO-R2
 ip address 10.0.1.1 255.255.255.0
 clock rate 64000
 no shutdown
interface Serial0/0/1
 description WAN2-TO-R4
 ip address 10.0.2.1 255.255.255.0
 clock rate 64000
 no shutdown
interface Serial0/1/0
 description WAN5-TO-R3
 ip address 10.0.5.1 255.255.255.0
 clock rate 64000
 no shutdown
exit
router ospf 1
 router-id 1.1.1.1
 network 10.0.1.0 0.0.0.255 area 0
 network 10.0.2.0 0.0.0.255 area 0
 network 10.0.5.0 0.0.0.255 area 0
 network 192.168.1.0 0.0.0.255 area 0
 passive-interface FastEthernet0/0
end
copy running-config startup-config

! R1: show controllers serial 0/0/0 | ping 10.0.1.2 | ping 192.168.4.11 source 192.168.1.1
! PC0: ipconfig | ping 192.168.1.1 | ping 192.168.4.11 | tracert 192.168.4.11
